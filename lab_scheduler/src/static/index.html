// Função corrigida para gerar PDF sem cortar os últimos dias async function generateSchedulePDF() { const scheduleTable = document.getElementById('scheduleTable'); if (!scheduleTable) { alert('Nenhuma escala carregada para gerar PDF.'); return; } try { // Criar um container temporário para o PDF const pdfContainer = document.createElement('div'); pdfContainer.style.position = 'absolute'; pdfContainer.style.left = '-9999px'; pdfContainer.style.top = '0'; pdfContainer.style.width = 'auto'; pdfContainer.style.minWidth = 'max-content'; pdfContainer.style.backgroundColor = 'white'; pdfContainer.style.padding = '20px'; // Criar uma cópia da tabela const tableClone = scheduleTable.cloneNode(true); // Aplicar estilos específicos para PDF na tabela clonada tableClone.style.width = 'auto'; tableClone.style.minWidth = 'max-content'; tableClone.style.tableLayout = 'auto'; tableClone.style.whiteSpace = 'nowrap'; tableClone.style.fontSize = '12px'; tableClone.style.fontFamily = 'Arial, sans-serif'; // Ajustar células da tabela const cells = tableClone.querySelectorAll('th, td'); cells.forEach(cell => { cell.style.padding = '8px'; cell.style.border = '1px solid #333'; cell.style.whiteSpace = 'nowrap'; cell.style.overflow = 'visible'; cell.style.minWidth = 'auto'; cell.style.maxWidth = 'none'; }); // Adicionar título ao container const title = document.createElement('h2'); title.textContent = 'Escala Semanal - Laboratórios ITV'; title.style.textAlign = 'center'; title.style.marginBottom = '20px'; title.style.fontSize = '18px'; title.style.fontFamily = 'Arial, sans-serif'; pdfContainer.appendChild(title); pdfContainer.appendChild(tableClone); // Adicionar ao DOM temporariamente document.body.appendChild(pdfContainer); // Aguardar um momento para renderização await new Promise(resolve => setTimeout(resolve, 100)); // Calcular dimensões necessárias const containerRect = pdfContainer.getBoundingClientRect(); const tableRect = tableClone.getBoundingClientRect(); // Definir configurações do html2canvas const canvasOptions = { allowTaint: true, useCORS: true, scale: 1, width: Math.max(containerRect.width, tableRect.width) + 40, height: containerRect.height + 40, scrollX: 0, scrollY: 0, backgroundColor: 'white' }; // Gerar canvas const canvas = await html2canvas(pdfContainer, canvasOptions); // Calcular orientação e tamanho do PDF const imgWidth = canvas.width; const imgHeight = canvas.height; const isLandscape = imgWidth > imgHeight; // Configurar jsPDF const orientation = isLandscape ? 'landscape' : 'portrait'; const unit = 'mm'; const format = 'a4'; const pdf = new jsPDF(orientation, unit, format); // Obter dimensões da página const pageWidth = pdf.internal.pageSize.getWidth(); const pageHeight = pdf.internal.pageSize.getHeight(); // Calcular escala para caber na página const scaleX = pageWidth / (imgWidth * 0.264583); // Converter px para mm const scaleY = pageHeight / (imgHeight * 0.264583); const scale = Math.min(scaleX, scaleY, 1); // Não aumentar, apenas diminuir se necessário const finalWidth = imgWidth * 0.264583 * scale; const finalHeight = imgHeight * 0.264583 * scale; // Centralizar na página const x = (pageWidth - finalWidth) / 2; const y = (pageHeight - finalHeight) / 2; // Converter canvas para imagem const imgData = canvas.toDataURL('image/png', 1.0); // Adicionar imagem ao PDF pdf.addImage(imgData, 'PNG', x, y, finalWidth, finalHeight); // Gerar nome do arquivo com data atual const currentDate = new Date(); const dateString = currentDate.toLocaleDateString('pt-BR').replace(/\//g, '-'); const weekSelector = document.getElementById('weekSelector'); const selectedWeek = weekSelector.value ? new Date(weekSelector.value + 'T00:00:00').toLocaleDateString('pt-BR').replace(/\//g, '-') : dateString; const fileName = `escala-laboratorios-${selectedWeek}.pdf`; // Salvar PDF pdf.save(fileName); // Remover container temporário document.body.removeChild(pdfContainer); console.log('PDF gerado com sucesso!'); } catch (error) { console.error('Erro ao gerar PDF:', error); alert('Erro ao gerar PDF. Verifique o console para mais detalhes.'); // Limpar container temporário se houver erro const tempContainer = document.querySelector('div[style*="position: absolute"][style*="left: -9999px"]'); if (tempContainer) { document.body.removeChild(tempContainer); } } } // Função auxiliar para garantir que as bibliotecas estão carregadas function ensureLibrariesLoaded() { return new Promise((resolve, reject) => { let attempts = 0; const maxAttempts = 50; const checkLibraries = () => { attempts++; if (typeof html2canvas !== 'undefined' && typeof jsPDF !== 'undefined') { resolve(); } else if (attempts >= maxAttempts) { reject(new Error('Bibliotecas html2canvas ou jsPDF não carregadas')); } else { setTimeout(checkLibraries, 100); } }; checkLibraries(); }); } // Event listener para o botão de gerar PDF document.addEventListener('DOMContentLoaded', function() { const generatePdfButton = document.getElementById('generatePdfButton'); if (generatePdfButton) { generatePdfButton.addEventListener('click', async function() { try { // Verificar se as bibliotecas estão carregadas await ensureLibrariesLoaded(); // Desabilitar botão durante geração this.disabled = true; this.textContent = 'Gerando PDF...'; // Gerar PDF await generateSchedulePDF(); // Reabilitar botão this.disabled = false; this.textContent = 'Gerar PDF da Escala'; } catch (error) { console.error('Erro:', error); alert('Erro ao gerar PDF: ' + error.message); // Reabilitar botão this.disabled = false; this.textContent = 'Gerar PDF da Escala'; } }); } }); // CSS adicional para garantir que a tabela seja renderizada corretamente const additionalStyles = ` `; // Adicionar os estilos ao head se ainda não existirem if (!document.querySelector('#pdf-fix-styles')) { const styleElement = document.createElement('style'); styleElement.id = 'pdf-fix-styles'; styleElement.innerHTML = additionalStyles.replace(/<\/?style>/g, ''); document.head.appendChild(styleElement); }
