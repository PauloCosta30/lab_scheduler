Correção: Erro de Geração de PDF

Sistema de Agendamentos - Filtro date_from_string não encontrado

Problema Identificado

Erro: "No filter named 'date_from_string'"
Causa: O template HTML está tentando usar um filtro personalizado do Jinja2 que não foi registrado no Flask.

Solução Implementada

Registrar o filtro personalizado no Flask e corrigir o template para processar datas corretamente.

# CORREÇÃO: Registrar filtro personalizado para templates
@bookings_bp.app_template_filter('date_from_string')
def date_from_string_filter(date_str):
    """Converte string de data para objeto datetime"""
    try:
        if isinstance(date_str, str):
            # Tenta diferentes formatos de data
            for fmt in ['%Y-%m-%d', '%d/%m/%Y', '%d-%m-%Y']:
                try:
                    return datetime.strptime(date_str, fmt).date()
                except ValueError:
                    continue
        elif isinstance(date_str, date):
            return date_str
        elif isinstance(date_str, datetime):
            return date_str.date()
        return None
    except Exception:
        return None

# CORREÇÃO: Função auxiliar para formatar nomes de dias da semana
@bookings_bp.app_template_filter('format_weekday')
def format_weekday_filter(date_obj):
    """Formata o nome do dia da semana em português"""
    if not date_obj:
        return ""
    
    weekdays = {
        0: 'Segunda-feira',
        1: 'Terça-feira', 
        2: 'Quarta-feira',
        3: 'Quinta-feira',
        4: 'Sexta-feira',
        5: 'Sábado',
        6: 'Domingo'
    }
    
    if isinstance(date_obj, str):
        date_obj = date_from_string_filter(date_obj)
    
    if date_obj:
        return weekdays.get(date_obj.weekday(), '')
    return ""

# Decorator para verificar se o PDF está disponível
def pdf_required(f):
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if not WEASYPRINT_AVAILABLE:
            return jsonify({"error": "Geração de PDF não está disponível"}), 503
        return f(*args, **kwargs)
    return decorated_function

@bookings_bp.route("/generate_schedule_pdf", methods=["POST"])
@pdf_required
def generate_schedule_pdf():
    try:
        data = request.get_json()
        start_date_str = data.get("start_date")
        end_date_str = data.get("end_date")

        if not start_date_str or not end_date_str:
            return jsonify({"error": "Datas de início e fim são obrigatórias"}), 400

        # Converter strings para objetos date
        start_date = datetime.strptime(start_date_str, "%Y-%m-%d").date()
        end_date = datetime.strptime(end_date_str, "%Y-%m-%d").date()

        # Buscar agendamentos
        bookings = db.session.query(Booking).join(Room).filter(
            Booking.booking_date >= start_date,
            Booking.booking_date <= end_date
        ).all()

        # CORREÇÃO: Organizar dados sem depender do filtro problemático
        schedule_data = defaultdict(lambda: defaultdict(list))
        user_observations = defaultdict(list)
        general_observations = []

        for booking in bookings:
            # Usar a data diretamente como objeto
            date_key = booking.booking_date.strftime('%Y-%m-%d')
            
            booking_info = {
                'user_name': booking.user_name,
                'room_name': booking.room.name,
                'coordinator_name': booking.coordinator_name,
                'observation': booking.observation
            }
            
            schedule_data[date_key][booking.period].append(booking_info)
            
            # Coletar observações
            if booking.observation:
                user_observations[booking.user_name].append({
                    'room': booking.room.name,
                    'date': booking.booking_date,
                    'period': booking.period,
                    'observation': booking.observation
                })
                
                general_observations.append({
                    'user_name': booking.user_name,
                    'coordinator_name': booking.coordinator_name,
                    'booking_date': booking.booking_date,
                    'observation': booking.observation
                })

        # Obter timestamp atual em Brasília
        now_brasilia = datetime.now(BRASILIA_TZ)

        # CORREÇÃO: Passar dados organizados para o template
        template_data = {
            'schedule_data': dict(schedule_data),
            'user_observations': dict(user_observations),
            'general_observations': general_observations,
            'start_date': start_date,
            'end_date': end_date,
            'generated_at': now_brasilia
        }

        # Renderizar template HTML
        html_content = render_template('schedule_pdf_template.html', **template_data)
        
        # Gerar PDF
        pdf = HTML(string=html_content).write_pdf()
        
        # Retornar PDF como resposta
        response = make_response(pdf)
        response.headers['Content-Type'] = 'application/pdf'
        response.headers['Content-Disposition'] = f'attachment; filename=agendamentos_{start_date_str}_a_{end_date_str}.pdf'
        
        return response

    except Exception as e:
        current_app.logger.error(f"Erro ao gerar PDF: {str(e)}")
        return jsonify({"error": f"Erro interno do servidor: {str(e)}"}), 500
